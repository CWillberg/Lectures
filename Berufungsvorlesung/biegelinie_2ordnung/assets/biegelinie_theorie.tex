% ---------- PAGE 2: exact curvature geometry (arc ds, d\varphi) ----------
\begin{tikzpicture}[>=stealth, line width=1pt, font=\large]

  \node[mainblue] at (-39:4.35) {$\mathrm ds$};
  \node[mainblue] at (-70:2.0) {$\rho$};

  \node[mainblue, font=\large, anchor=west] at (6.4,1.0) {$\rho = \dfrac{\mathrm ds}{\mathrm d\varphi}$};

  \node[mainblue, font=\large, anchor=west] at (6.4,-1.0) {$\dfrac{1}{\rho} = \dfrac{w''(x)}{\left(1+w'(x)^2\right)^{3/2}}$};

  \node[mainblue, font=\large, anchor=west] at (6.4,-3.0) {Technische Biegelehre:};
  \node[mainblue, font=\large, anchor=west] at (6.4,-4.0) {$w' \ll 1 \ \Rightarrow\ \dfrac{1}{\rho}\approx w''(x)$};

% ---------- PAGE 3: differential beam element dx (equilibrium) ----------
\begin{tikzpicture}[>=stealth, line width=1pt, font=\large]
  % small x,z coordinate indicator
  \draw[->, mainblue, line width=0.8pt] (-1.5,1.9) -- (-0.9,1.9) node[right] {$x$};
  \draw[->, mainblue, line width=0.8pt] (-1.5,1.9) -- (-1.5,1.3) node[below] {$z$};

  % beam element box
  \draw[mainblue, line width=1.4pt] (0,0) rectangle (4,0.8);

  % distributed load q(x): arbitrary, closed-form function above the element, confined to [0,4]
  \draw[loadcolor, line width=1.3pt]
    plot[domain=0:4,samples=60] (\x,{1.7+0.35*sin(45*\x)});
  \foreach \x in {0.4,1.2,...,3.6} {
    \draw[loadcolor, ->, line width=1pt] (\x,{1.7+0.35*sin(45*\x)}) -- (\x,0.85);
  }
  \node[loadcolor] at (2.0,2.45) {$q(x)$};

  % left cut face, left to right: F_L (arrow pointing further left), M_b, Balken
  \draw[normalcolor, ->, line width=1.3pt] (-0.8,0.4) -- (-1.8,0.4) node[left] {$F_L$};
  \draw[mainblue, ->, line width=1.3pt] let \p1=($(-0.4,0.4)+(270:0.35)$) in (\p1) arc (270:90:0.35);
  \node[mainblue] at (-0.4,0.95) {$M_b$};
  \draw[shearcolor, ->, line width=1.3pt] (0,0.4) -- (0,1.7) node[above] {$F_q$};

  % right cut face, left to right: Balken, M_b+dM_b, F_L+dF_L (arrow pointing further right)
  \draw[shearcolor, ->, line width=1.3pt] (4,0.4) -- (4,-1.3) node[below] {$F_q+\mathrm dF_q$};
  \draw[mainblue, ->, line width=1.3pt] let \p1=($(4.4,0.4)+(-90:0.35)$) in (\p1) arc (-90:90:0.35);
  \node[mainblue] at (4.4,0.95) {$M_b+\mathrm dM_b$};
  \draw[normalcolor, ->, line width=1.3pt] (4.8,0.4) -- (5.8,0.4) node[right] {$F_L+\mathrm dF_L$};

  \node[accent] at (2.0,-1.9) {$\mathrm dx$};

  \node[mainblue, font=\large] at (2.0,-3.0) {$\dfrac{\mathrm dF_q}{\mathrm dx} = -q(x) \qquad \dfrac{\mathrm dM_b}{\mathrm dx} = F_q(x)$};
\end{tikzpicture}

% ---------- PAGE 4: fiber strain geometry (plane sections remain plane) ----------
\begin{tikzpicture}[>=stealth, line width=1pt, font=\large]

  \draw[mainblue, line width=1.8pt] (-70:4) arc (-70:-20:4);

  \node[mainblue, anchor=south west] at (-22:5.25) {$\mathrm ds+z\,\mathrm d\varphi$};

  \draw[shearcolor, <->, line width=0.9pt] (-20:4.08) -- (-20:4.82);
  \node[shearcolor, anchor=north west] at (-20:4.45) {$z$};

  \node[accent] at (-45:1.3) {$\mathrm d\varphi$};

  \node[mainblue, font=\large, anchor=west] at (7.3,0.0) {Faser im Abstand $z$: $\mathrm ds+z\,\mathrm d\varphi=(\rho+z)\,\mathrm d\varphi$};
  \node[mainblue, font=\large, anchor=west] at (7.3,-1.6) {Dehnung: $\varepsilon(x,z)=\dfrac{(\mathrm ds+z\,\mathrm d\varphi)-\mathrm ds}{\mathrm ds} = \dfrac{z\,\mathrm d\varphi}{\rho\,\mathrm d\varphi}$};
  \node[mainblue, font=\large, anchor=west] at (7.3,-3.0) {$\boxed{\varepsilon(x,z) = \dfrac{z}{\rho(x)}}$};
\end{tikzpicture}
